const calculateAgeString = (dob) => {
    if (!dob) return '';
    const birthDate = dayjs(dob);
    const today = dayjs();
    const years = today.diff(birthDate, 'year');
    let tempDate = birthDate.add(years, 'year');
    const months = today.diff(tempDate, 'month');
    tempDate = tempDate.add(months, 'month');
    const days = today.diff(tempDate, 'day');
    return `${years} tuổi, ${months} tháng, ${days} ngày`;
};

    const onChange = (key) => setActiveKey(key);
    const onEdit = (targetKey, action) => {
        if (action === 'add') {
             const newActiveKey = `donhang-${Date.now()}`;
             setPanes([...panes, { title: `Đơn hàng ${panes.length + 1}`, key: newActiveKey, cart: [], customer: null, predictedDays: 0 }]);
             setActiveKey(newActiveKey);
        } else {
             let newActiveKey = activeKey;
             let lastIndex = -1;
             panes.forEach((pane, i) => { if (pane.key === targetKey) { lastIndex = i - 1; } });

             if (newPanes.length && newActiveKey === targetKey) {
                 newActiveKey = lastIndex >= 0 ? newPanes[lastIndex].key : newPanes[0].key;
             }
             if (newPanes.length === 0) {
                 const newKey = `donhang-${Date.now()}`;

             }

                    const paneIndex = panes.findIndex(pane => pane.key === key);

                 ...item,
                 key: `item-${Date.now()}-${index}`,

             const newCart = cart.map(item => item.key === key ? { ...item, [field]: value } : item);

                    key: `${product.id}-${Date.now()}`,
                    id: product.id,
                    name: product.name,
                    quantity: item.quantity,
                    price: product.price,
                    unit: product.unit,
                    hdsd: product.hdsd,

            });

                            dataSource={cart}

                                        <Col flex="auto">

                                        </Col>

                                        <Col style={{width: 100}}><InputNumber min={1} value={item.quantity} onChange={(val) => handleCartItemChange(item.key, 'quantity', val)}/></Col>

                            </Row>

                    </Card>
                </Col>

        )
    }

    return (
        <ConfigProvider locale={viVN} theme={{ token: { colorPrimary: '#003a78' }}}>
            <Layout style={{ height: '100vh', background: '#f5f5f5' }}>
                <Content style={{ paddingTop: '12px', paddingLeft: '24px' }}>
                    <Tabs type="editable-card" onChange={onChange} activeKey={activeKey} onEdit={onEdit} hideAdd={panes.length >= 5}>
                        {panes.map((pane) => (
                            <TabPane tab={pane.title} key={pane.key} closable={panes.length > 1}>
                               <PosTabContent pane={pane} updateData={updatePaneData} />
                            </TabPane>
                        ))}
                    </Tabs>
                </Content>
            </Layout>
        </ConfigProvider>
    );
};


export default PosInterfaceSmartPrescription;
